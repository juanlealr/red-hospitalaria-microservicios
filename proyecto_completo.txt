package co.edu.uptc.farmacia_service.dominio;

import org.junit.jupiter.api.Test;

import java.time.LocalDateTime;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertNull;
import static org.junit.jupiter.api.Assertions.assertThrows;

class SolicitudMedicamentoTest {

    private SolicitudMedicamento solicitudRecibida() {
        return new SolicitudMedicamento(
                "sol-1", "pac-1", "med-1", "Ibuprofeno 400 mg", LocalDateTime.of(2026, 9, 29, 8, 0));
    }

    @Test
    void iniciaRecibidaYSinCobertura() {
        var solicitud = solicitudRecibida();

        assertEquals(EstadoSolicitudMedicamento.RECIBIDA, solicitud.estado());
        assertNull(solicitud.cobertura());
        assertNull(solicitud.fechaDespacho());
    }

    @Test
    void noSePuedeDespacharSinCoberturaVerificada() {
        var solicitud = solicitudRecibida();

        assertThrows(IllegalStateException.class, () -> solicitud.despachar(LocalDateTime.now()));
    }

    @Test
    void registraCoberturaYPasaACoberturaVerificada() {
        var solicitud = solicitudRecibida();

        solicitud.registrarCobertura(new Cobertura(80));

        assertEquals(EstadoSolicitudMedicamento.COBERTURA_VERIFICADA, solicitud.estado());
        assertEquals(new Cobertura(80), solicitud.cobertura());
    }

    @Test
    void despachaDespuesDeVerificarLaCobertura() {
        var solicitud = solicitudRecibida();
        solicitud.registrarCobertura(new Cobertura(80));
        var fechaDespacho = LocalDateTime.of(2026, 9, 29, 9, 30);

        solicitud.despachar(fechaDespacho);

        assertEquals(EstadoSolicitudMedicamento.DESPACHADA, solicitud.estado());
        assertEquals(fechaDespacho, solicitud.fechaDespacho());
    }

    @Test
    void noReemplazaLaCoberturaYaVerificada() {
        var solicitud = solicitudRecibida();
        solicitud.registrarCobertura(new Cobertura(80));

        assertThrows(IllegalStateException.class, () -> solicitud.registrarCobertura(new Cobertura(90)));
    }

    @Test
    void noDespachaDosVeces() {
        var solicitud = solicitudRecibida();
        solicitud.registrarCobertura(new Cobertura(80));
        solicitud.despachar(LocalDateTime.now());

        assertThrows(IllegalStateException.class, () -> solicitud.despachar(LocalDateTime.now()));
    }

    @Test
    void reconstruyeUnaSolicitudDespachadaDesdePersistencia() {
        var fechaSolicitud = LocalDateTime.of(2026, 9, 29, 8, 0);
        var fechaDespacho = LocalDateTime.of(2026, 9, 29, 9, 30);

        var solicitud = SolicitudMedicamento.reconstruir(
                "sol-1", "pac-1", "med-1", "Ibuprofeno 400 mg", fechaSolicitud,
                EstadoSolicitudMedicamento.DESPACHADA, new Cobertura(80), fechaDespacho);

        assertEquals(EstadoSolicitudMedicamento.DESPACHADA, solicitud.estado());
        assertEquals(new Cobertura(80), solicitud.cobertura());
        assertEquals(fechaDespacho, solicitud.fechaDespacho());
    }

    @Test
    void reconstruirRechazaCombinacionesIncoherentes() {
        var fecha = LocalDateTime.of(2026, 9, 29, 8, 0);

        assertThrows(IllegalStateException.class, () -> SolicitudMedicamento.reconstruir(
                "sol-1", "pac-1", "med-1", "Ibuprofeno 400 mg", fecha,
                EstadoSolicitudMedicamento.RECIBIDA, new Cobertura(80), null));

        assertThrows(IllegalStateException.class, () -> SolicitudMedicamento.reconstruir(
                "sol-1", "pac-1", "med-1", "Ibuprofeno 400 mg", fecha,
                EstadoSolicitudMedicamento.COBERTURA_VERIFICADA, null, null));

        assertThrows(IllegalStateException.class, () -> SolicitudMedicamento.reconstruir(
                "sol-1", "pac-1", "med-1", "Ibuprofeno 400 mg", fecha,
                EstadoSolicitudMedicamento.DESPACHADA, new Cobertura(80), null));
    }
}

import static org.junit.jupiter.api.Assertions.assertNotNull;
import static org.junit.jupiter.api.Assertions.assertThrows;
import static org.junit.jupiter.api.Assertions.assertTrue;

class SolicitudMedicamentoServiceTest {

    private final FakeRepositorioSolicitudes repositorio = new FakeRepositorioSolicitudes();
    private final FakePublicadorEventos publicador = new FakePublicadorEventos();
    private final SolicitudMedicamentoService servicio = new SolicitudMedicamentoService(
            repositorio,
            new SolicitudMedicamentoFactory(),
            new VerificacionCoberturaService(pacienteId -> new Cobertura(80)),
            new RegistroSolicitudTransaccional(repositorio, publicador));

    @Test
    void recibeGuardaYPublicaMedicamentoSolicitado() {
        var solicitud = servicio.recibir("pac-1", "med-1", "Ibuprofeno 400 mg");

        assertNotNull(solicitud.id());
        assertEquals(EstadoSolicitudMedicamento.RECIBIDA, solicitud.estado());
        assertTrue(repositorio.buscarPorId(solicitud.id()).isPresent());
        assertEquals(List.of(EventosFarmacia.MEDICAMENTO_SOLICITADO), publicador.tipos);
    }

    @Test
    void verificaLaCoberturaYPublicaCoberturaVerificada() {
        var solicitud = servicio.recibir("pac-1", "med-1", "Ibuprofeno 400 mg");

        var actualizada = servicio.verificarCobertura(solicitud.id());

        assertEquals(EstadoSolicitudMedicamento.COBERTURA_VERIFICADA, actualizada.estado());
        assertEquals(new Cobertura(80), actualizada.cobertura());
        assertEquals(EventosFarmacia.COBERTURA_VERIFICADA, publicador.tipos.get(1));
    }

    @Test
    void despachaDespuesDeVerificarLaCobertura() {
        var solicitud = servicio.recibir("pac-1", "med-1", "Ibuprofeno 400 mg");
        servicio.verificarCobertura(solicitud.id());

        var despachada = servicio.despachar(solicitud.id());

        assertEquals(EstadoSolicitudMedicamento.DESPACHADA, despachada.estado());
        assertNotNull(despachada.fechaDespacho());
        assertEquals(EventosFarmacia.MEDICAMENTO_DESPACHADO, publicador.tipos.get(2));
    }

    @Test
    void noDespachaSinCoberturaVerificada() {
        var solicitud = servicio.recibir("pac-1", "med-1", "Ibuprofeno 400 mg");

        assertThrows(IllegalStateException.class, () -> servicio.despachar(solicitud.id()));
        assertEquals(List.of(EventosFarmacia.MEDICAMENTO_SOLICITADO), publicador.tipos);
    }

    @Test
    void siLaAseguradoraFallaLaSolicitudQuedaRecibidaYSinEvento() {
        var repositorioLocal = new FakeRepositorioSolicitudes();
        var publicadorLocal = new FakePublicadorEventos();
        var servicioConFallo = new SolicitudMedicamentoService(
                repositorioLocal,
                new SolicitudMedicamentoFactory(),
                new VerificacionCoberturaService(pacienteId -> {
                    throw new IllegalStateException("aseguradora caída");
                }),
                new RegistroSolicitudTransaccional(repositorioLocal, publicadorLocal));

        var solicitud = servicioConFallo.recibir("pac-1", "med-1", "Ibuprofeno 400 mg");

        assertThrows(IllegalStateException.class, () -> servicioConFallo.verificarCobertura(solicitud.id()));

        var sinCambios = repositorioLocal.buscarPorId(solicitud.id()).orElseThrow();
        assertEquals(EstadoSolicitudMedicamento.RECIBIDA, sinCambios.estado());
        assertEquals(List.of(EventosFarmacia.MEDICAMENTO_SOLICITADO), publicadorLocal.tipos);
    }

    @Test
    void fallaSiLaSolicitudNoExiste() {
        assertThrows(NoSuchElementException.class, () -> servicio.buscarPorId("no-existe"));
        assertThrows(NoSuchElementException.class, () -> servicio.verificarCobertura("no-existe"));
    }

    @Test
    void listaTodasLasSolicitudes() {
        servicio.recibir("pac-1", "med-1", "Ibuprofeno 400 mg");
        servicio.recibir("pac-2", "med-2", "Paracetamol 500 mg");

        assertEquals(2, servicio.listarTodas().size());
    }

    private static class FakeRepositorioSolicitudes implements RepositorioSolicitudesMedicamento {

        private final Map<String, SolicitudMedicamento> solicitudes = new HashMap<>();

        @Override
        public SolicitudMedicamento guardar(SolicitudMedicamento solicitud) {
            solicitudes.put(solicitud.id(), solicitud);
            return solicitud;
        }

        @Override
        public Optional<SolicitudMedicamento> buscarPorId(String id) {
            return Optional.ofNullable(solicitudes.get(id));
        }

        @Override
        public List<SolicitudMedicamento> listarTodas() {
            return new ArrayList<>(solicitudes.values());
        }
    }

    private static class FakePublicadorEventos implements PublicadorEventosSolicitud {

        private final List<String> tipos = new ArrayList<>();

        @Override
        public void publicar(String tipoEvento, SolicitudMedicamento solicitud) {
            tipos.add(tipoEvento);
        }
    }
}

package co.edu.uptc.farmacia_service.dominio;

import java.time.LocalDateTime;

/**
 * Raíz del Agregado: SolicitudMedicamento.
 *
 * Es el único punto de entrada del subdominio "Farmacia y cobertura". Todo
 * lo que cambia junto y de forma consistente vive aquí: el estado de la
 * solicitud (RECIBIDA -> COBERTURA_VERIFICADA -> DESPACHADA) y la cobertura
 * verificada.
 *
 * Referencias a otros subdominios: siempre por id, nunca por objeto
 * completo (pacienteId, medicoId). El núcleo no importa Spring, JPA ni
 * ningún cliente HTTP: la persistencia y la aseguradora entran por puertos.
 */
public class SolicitudMedicamento {

    private final String id;

    private final String pacienteId;

    private final String medicoId;

    private final String medicamento;

    private final LocalDateTime fechaSolicitud;

    private EstadoSolicitudMedicamento estado;

    private Cobertura cobertura;

    private LocalDateTime fechaDespacho;

    /**
     * Solo debe llamarse desde SolicitudMedicamentoFactory (o desde
     * {@link #reconstruir} al rehidratar desde persistencia).
     */
    public SolicitudMedicamento(
            String id,
            String pacienteId,
            String medicoId,
            String medicamento,
            LocalDateTime fechaSolicitud) {
        this.id = id;
        this.pacienteId = pacienteId;
        this.medicoId = medicoId;
        this.medicamento = medicamento;
        this.fechaSolicitud = fechaSolicitud;
        this.estado = EstadoSolicitudMedicamento.RECIBIDA;
    }

    /**
     * Rehidrata el Agregado desde persistencia sin volver a ejecutar las
     * transiciones de estado (que fijarían fechas nuevas). Uso exclusivo de
     * los adaptadores de salida; valida que la combinación estado/cobertura/
     * fecha sea coherente para no reconstruir una solicitud imposible.
     */
    public static SolicitudMedicamento reconstruir(
            String id,
            String pacienteId,
            String medicoId,
            String medicamento,
            LocalDateTime fechaSolicitud,
            EstadoSolicitudMedicamento estado,
            Cobertura cobertura,
            LocalDateTime fechaDespacho) {
        exigirTexto(id, "El id de la solicitud es obligatorio.");
        exigirTexto(pacienteId, "El id del paciente es obligatorio.");
        exigirTexto(medicoId, "El id del médico es obligatorio.");
        exigirTexto(medicamento, "El medicamento es obligatorio.");
        if (fechaSolicitud == null) {
            throw new IllegalArgumentException("La fecha de solicitud es obligatoria.");
        }
        if (estado == null) {
            throw new IllegalArgumentException("El estado de la solicitud es obligatorio.");
        }
        switch (estado) {
            case RECIBIDA -> {
                if (cobertura != null || fechaDespacho != null) {
                    throw new IllegalStateException("Una solicitud RECIBIDA no puede tener cobertura ni fecha de despacho.");
                }
            }
            case COBERTURA_VERIFICADA -> {
                if (cobertura == null) {
                    throw new IllegalStateException("Una solicitud con cobertura verificada debe tener cobertura.");
                }
                if (fechaDespacho != null) {
                    throw new IllegalStateException("Una solicitud no despachada no puede tener fecha de despacho.");
                }
            }
            case DESPACHADA -> {
                if (cobertura == null) {
                    throw new IllegalStateException("Una solicitud DESPACHADA debe tener cobertura verificada.");
                }
                if (fechaDespacho == null) {
                    throw new IllegalStateException("Una solicitud DESPACHADA debe tener fecha de despacho.");
                }
            }
        }

        SolicitudMedicamento solicitud = new SolicitudMedicamento(id, pacienteId, medicoId, medicamento, fechaSolicitud);
        solicitud.estado = estado;
        solicitud.cobertura = cobertura;
        solicitud.fechaDespacho = fechaDespacho;
        return solicitud;
    }

    /**
     * Registra la cobertura que confirmó la aseguradora externa (HU-07).
     * Solo tiene sentido sobre una solicitud recién recibida: una vez
     * verificada, la cobertura no se reemplaza.
     */
    public void registrarCobertura(Cobertura cobertura) {
        if (cobertura == null) {
            throw new IllegalArgumentException("La cobertura verificada es obligatoria.");
        }
        if (estado != EstadoSolicitudMedicamento.RECIBIDA) {
            throw new IllegalStateException(
                    "Solo se registra la cobertura de una solicitud recibida (estado actual: " + estado + ").");
        }
        this.cobertura = cobertura;
        this.estado = EstadoSolicitudMedicamento.COBERTURA_VERIFICADA;
    }

    /**
     * Despacho del medicamento al paciente (HU-06). Exige que la cobertura
     * ya haya sido verificada.
     */
    public void despachar(LocalDateTime fecha) {
        if (fecha == null) {
            throw new IllegalArgumentException("El despacho debe tener fecha.");
        }
        if (estado == EstadoSolicitudMedicamento.DESPACHADA) {
            throw new IllegalStateException("La solicitud ya fue despachada.");
        }
        if (estado != EstadoSolicitudMedicamento.COBERTURA_VERIFICADA) {
            throw new IllegalStateException(
                    "No se puede despachar sin cobertura verificada (estado actual: " + estado + ").");
        }
        this.estado = EstadoSolicitudMedicamento.DESPACHADA;
        this.fechaDespacho = fecha;
    }

    private static void exigirTexto(String valor, String mensaje) {
        if (valor == null || valor.isBlank()) {
            throw new IllegalArgumentException(mensaje);
        }
    }

    public String id() {
        return id;
    }

    public String pacienteId() {
        return pacienteId;
    }

    public String medicoId() {
        return medicoId;
    }

    public String medicamento() {
        return medicamento;
    }

    public LocalDateTime fechaSolicitud() {
        return fechaSolicitud;
    }

    public EstadoSolicitudMedicamento estado() {
        return estado;
    }

    public Cobertura cobertura() {
        return cobertura;
    }

    public LocalDateTime fechaDespacho() {
        return fechaDespacho;
    }
}

import co.edu.uptc.farmacia_service.dominio.SolicitudMedicamento;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

/**
 * Frontera transaccional de escritura del subdominio.
 *
 * Existe separada del caso de uso por dos razones concretas:
 * 1. La llamada a la aseguradora es de red y no debe ejecutarse dentro de
 *    una transacción de base de datos (no se retiene una conexión mientras
 *    se espera a un tercero).
 * 2. El cambio de estado y el evento en el outbox tienen que confirmarse
 *    en la misma transacción — es la garantía del patrón Transactional
 *    Outbox.
 */
@Service
public class RegistroSolicitudTransaccional {

    private final RepositorioSolicitudesMedicamento repositorio;
    private final PublicadorEventosSolicitud publicador;

    public RegistroSolicitudTransaccional(
            RepositorioSolicitudesMedicamento repositorio,
            PublicadorEventosSolicitud publicador) {
        this.repositorio = repositorio;
        this.publicador = publicador;
    }

    @Transactional
    public SolicitudMedicamento guardar(SolicitudMedicamento solicitud, String tipoEvento) {
        SolicitudMedicamento persistida = repositorio.guardar(solicitud);
        publicador.publicar(tipoEvento, persistida);
        return persistida;
    }
}

/**
 * Factory (paso 5 del taller de DDD): mueve aquí la validación y
 * construcción de la raíz del Agregado.
 *
 * @Component: Spring la gestiona como bean, y nada fuera de esta clase
 * debería construir una SolicitudMedicamento directamente.
 *
 * Cubre HU-06 ("recibir ... solicitudes de medicamentos") y refuerza la

@Component

                    "Se requiere la solicitud médica asociada (id del médico): un medicamento no se despacha sin ella.");
        }
        if (medicamento == null || medicamento.isBlank()) {
            throw new IllegalArgumentException("Se requiere el medicamento solicitado.");

/**
 * Servicio de Dominio (paso 3): VerificacionCoberturaService.
 *
 * Cubre HU-07: "verificar la cobertura del paciente ante una aseguradora
 * externa, para saber qué está cubierto". La operación no pertenece
 * naturalmente a SolicitudMedicamento (que no debería saber cómo se
 * consulta la aseguradora) ni a Cobertura (que no conoce la solicitud):
 * necesita a ambos, por eso es un Servicio de Dominio.
 *
 * La aseguradora entra por el puerto ConsultaCoberturaExterna — nunca como
 * una dependencia concreta. Si el sistema externo falla, la excepción sube
 * y la solicitud no cambia de estado: la operación clínica interna
 * (recibir, despachar solicitudes ya verificadas, listar) sigue disponible.
 */
@Service
public class VerificacionCoberturaService {

    private final ConsultaCoberturaExterna consultaCoberturaExterna;

    public VerificacionCoberturaService(ConsultaCoberturaExterna consultaCoberturaExterna) {
        this.consultaCoberturaExterna = consultaCoberturaExterna;
    }

    /**
     * Consulta la cobertura del paciente en la aseguradora externa y la
     * registra en la solicitud: a partir de este punto la solicitud queda
     * apta para despacho (evento CoberturaVerificada del Event Storming).
     */
    public Cobertura verificar(SolicitudMedicamento solicitud) {
        if (solicitud == null) {
            throw new IllegalArgumentException("La solicitud de medicamento es obligatoria.");
        }
        Cobertura cobertura = consultaCoberturaExterna.consultar(solicitud.pacienteId());
        solicitud.registrarCobertura(cobertura);
        return cobertura;
    }
}

package co.edu.uptc.farmacia_service.aplicacion;

import co.edu.uptc.farmacia_service.dominio.SolicitudMedicamento;
import org.springframework.stereotype.Service;

import java.time.LocalDateTime;
import java.util.List;
import java.util.NoSuchElementException;

/**
 * Caso de uso concreto de Farmacia y cobertura.
 *
 * Orquesta el Agregado, la Factory, el Servicio de Dominio de cobertura y
 * los puertos secundarios. No conoce HTTP, ni JPA, ni la tecnología
 * concreta de la aseguradora: todo entra por puertos.
 */
@Service
public class SolicitudMedicamentoService implements SolicitudMedicamentoUseCase {

    private final RepositorioSolicitudesMedicamento repositorio;
    private final SolicitudMedicamentoFactory factory;
    private final VerificacionCoberturaService verificacionCoberturaService;
    private final RegistroSolicitudTransaccional registro;

    public SolicitudMedicamentoService(
            RepositorioSolicitudesMedicamento repositorio,
            SolicitudMedicamentoFactory factory,
            VerificacionCoberturaService verificacionCoberturaService,
            RegistroSolicitudTransaccional registro) {
        this.repositorio = repositorio;
        this.factory = factory;
        this.verificacionCoberturaService = verificacionCoberturaService;
        this.registro = registro;
    }

    @Override
    public SolicitudMedicamento recibir(String pacienteId, String medicoId, String medicamento) {
        SolicitudMedicamento solicitud = factory.recibirSolicitud(pacienteId, medicoId, medicamento);
        return registro.guardar(solicitud, EventosFarmacia.MEDICAMENTO_SOLICITADO);
    }

    @Override
    public SolicitudMedicamento verificarCobertura(String solicitudId) {
        SolicitudMedicamento solicitud = buscarPorId(solicitudId);

        // Llamada de red fuera de la transacción: si la aseguradora falla,
        // la solicitud queda intacta en RECIBIDA.
        verificacionCoberturaService.verificar(solicitud);

        return registro.guardar(solicitud, EventosFarmacia.COBERTURA_VERIFICADA);
    }

    @Override
    public SolicitudMedicamento despachar(String solicitudId) {
        SolicitudMedicamento solicitud = buscarPorId(solicitudId);
        solicitud.despachar(LocalDateTime.now());
        return registro.guardar(solicitud, EventosFarmacia.MEDICAMENTO_DESPACHADO);
    }

    @Override
    public SolicitudMedicamento buscarPorId(String solicitudId) {
        if (solicitudId == null || solicitudId.isBlank()) {
            throw new IllegalArgumentException("El id de la solicitud es obligatorio.");
        }
        return repositorio.buscarPorId(solicitudId)
                .orElseThrow(() -> new NoSuchElementException(

import co.edu.uptc.farmacia_service.aplicacion.PublicadorEventosSolicitud;
import co.edu.uptc.farmacia_service.dominio.SolicitudMedicamento;
import co.edu.uptc.farmacia_service.infraestructura.salida.persistencia.OutboxEventEntity;
import co.edu.uptc.farmacia_service.infraestructura.salida.persistencia.OutboxEventRepository;
import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.stereotype.Component;

import java.time.LocalDateTime;
import java.util.UUID;

/**
 * Adaptador secundario del puerto de publicación: convierte el hecho de
 * dominio en una fila de outbox dentro de la transacción en curso. La
 * entrega real a Kafka la hace OutboxEventPublisher.
 */
@Component
public class PublicadorEventosOutboxAdapter implements PublicadorEventosSolicitud {

    private final OutboxEventRepository outboxRepository;
    private final ObjectMapper objectMapper;

    public PublicadorEventosOutboxAdapter(OutboxEventRepository outboxRepository, ObjectMapper objectMapper) {
        this.outboxRepository = outboxRepository;
        this.objectMapper = objectMapper;
    }

    @Override
    public void publicar(String tipoEvento, SolicitudMedicamento solicitud) {
        String eventoId = UUID.randomUUID().toString();
        LocalDateTime ocurridoEn = LocalDateTime.now();

        EventoSolicitudPayload payload = new EventoSolicitudPayload(
                eventoId,
                tipoEvento,
                ocurridoEn,
                solicitud.id(),
                solicitud.pacienteId(),
                solicitud.medicoId(),
                solicitud.medicamento(),
                solicitud.estado().name(),
                solicitud.cobertura() == null ? null : solicitud.cobertura().porcentaje(),
                solicitud.fechaSolicitud(),
                solicitud.fechaDespacho());

        outboxRepository.save(new OutboxEventEntity(
                eventoId,
                solicitud.id(),
                tipoEvento,
                serializar(payload),
                ocurridoEn));
    }

    private String serializar(EventoSolicitudPayload payload) {
        try {
            return objectMapper.writeValueAsString(payload);
        } catch (JsonProcessingException e) {
            throw new IllegalStateException("No se pudo serializar el evento " + payload.tipo(), e);
        }
    }
}

package co.edu.uptc.farmacia_service.infraestructura.salida.mensajeria;

import co.edu.uptc.farmacia_service.infraestructura.salida.persistencia.OutboxEventEntity;
import co.edu.uptc.farmacia_service.infraestructura.salida.persistencia.OutboxEventRepository;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Transactional;

import java.util.concurrent.TimeUnit;

/**
 * Publica en Kafka los eventos pendientes de la tabla outbox.
 *
 * Solo marca el evento como publicado cuando el bróker confirma; si falla,
 * lo deja pendiente y corta el lote para preservar el orden por agregado.
 */
@Component
public class OutboxEventPublisher {

    private static final Logger log = LoggerFactory.getLogger(OutboxEventPublisher.class);

    private final OutboxEventRepository outboxRepository;
    private final KafkaTemplate<String, String> kafkaTemplate;
    private final String topic;

    public OutboxEventPublisher(
            OutboxEventRepository outboxRepository,
            KafkaTemplate<String, String> kafkaTemplate,
            @Value("${farmacia.kafka.topic-solicitudes:farmacia.solicitudes}") String topic) {
        this.outboxRepository = outboxRepository;
        this.kafkaTemplate = kafkaTemplate;
        this.topic = topic;
    }

    @Scheduled(fixedDelayString = "${farmacia.outbox.intervalo-ms:2000}")
    @Transactional
    public void publicarPendientes() {
        for (OutboxEventEntity evento : outboxRepository.buscarPendientesParaPublicar()) {
            try {
                kafkaTemplate.send(topic, evento.getClave(), evento.getPayload())
                        .get(5, TimeUnit.SECONDS);
                evento.marcarPublicado();
                log.info("Outbox: evento {} ({}) publicado en {}", evento.getId(), evento.getTipo(), topic);
            } catch (Exception e) {
                log.warn("Outbox: no se pudo publicar el evento {}. Se reintentará en el siguiente ciclo",
                        evento.getId());
                return;
            }
        }
    }
}

package co.edu.uptc.farmacia_service.infraestructura.salida.aseguradora;

import co.edu.uptc.farmacia_service.aplicacion.ConsultaCoberturaExterna;
import co.edu.uptc.farmacia_service.compartido.ServicioExternoNoDisponibleException;
import co.edu.uptc.farmacia_service.dominio.Cobertura;
import io.github.resilience4j.circuitbreaker.CallNotPermittedException;
import io.github.resilience4j.circuitbreaker.CircuitBreaker;
import io.github.resilience4j.circuitbreaker.CircuitBreakerConfig;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.context.annotation.Profile;
import org.springframework.http.client.SimpleClientHttpRequestFactory;
import org.springframework.stereotype.Component;
import org.springframework.web.client.RestClient;
import org.springframework.web.client.RestClientException;

import java.time.Duration;

/**
 * Adaptador secundario real del puerto ConsultaCoberturaExterna: llama por
 * REST a la aseguradora externa.
 *
 * Aplica el patrón de la Lección 6 para no bloquear la operación clínica:
 * timeouts acotados y Circuit Breaker (Resilience4j) — si el externo falla
 * repetidamente, las llamadas siguientes fallan de inmediato con 503 en vez
 * de agotar hilos esperando.
 *
 * Se activa con el perfil "aseguradora-rest".
 */
@Component
@Profile("aseguradora-rest")
public class CoberturaAseguradoraRestAdapter implements ConsultaCoberturaExterna {

    private static final Logger log = LoggerFactory.getLogger(CoberturaAseguradoraRestAdapter.class);

    private final RestClient restClient;
    private final CircuitBreaker circuitBreaker;

    public CoberturaAseguradoraRestAdapter(AseguradoraProperties properties) {
        SimpleClientHttpRequestFactory requestFactory = new SimpleClientHttpRequestFactory();
        requestFactory.setConnectTimeout(Duration.ofMillis(properties.connectTimeoutMs()));
        requestFactory.setReadTimeout(Duration.ofMillis(properties.readTimeoutMs()));

        this.restClient = RestClient.builder()
                .baseUrl(properties.url())
                .requestFactory(requestFactory)
                .build();

        this.circuitBreaker = CircuitBreaker.of("aseguradora",
                CircuitBreakerConfig.custom()
                        .slidingWindowSize(10)
                        .minimumNumberOfCalls(5)
                        .failureRateThreshold(50)
                        .waitDurationInOpenState(Duration.ofSeconds(10))
                        .build());
        this.circuitBreaker.getEventPublisher().onStateTransition(
                e -> log.warn("Circuit Breaker aseguradora: {}", e.getStateTransition()));
    }

    @Override
    public Cobertura consultar(String pacienteId) {
        try {
            return circuitBreaker.executeSupplier(() -> {
                CoberturaRespuesta respuesta = restClient.get()
                        .uri("/cobertura/{pacienteId}", pacienteId)
                        .retrieve()
                        .body(CoberturaRespuesta.class);

                if (respuesta == null || respuesta.porcentaje() == null) {
                    throw new ServicioExternoNoDisponibleException(
                            "La aseguradora respondió sin porcentaje de cobertura", null);
                }
                return new Cobertura(respuesta.porcentaje());
            });
        } catch (CallNotPermittedException e) {
            log.warn("Circuit Breaker ABIERTO: aseguradora no disponible");
            throw new ServicioExternoNoDisponibleException(
                    "La aseguradora no está disponible (circuito abierto)", e);
        } catch (RestClientException e) {
            throw new ServicioExternoNoDisponibleException(
                    "No se pudo consultar la cobertura en la aseguradora", e);
        }
    }
}

package co.edu.uptc.farmacia_service.infraestructura.entrada.rest;

import co.edu.uptc.farmacia_service.compartido.ServicioExternoNoDisponibleException;
import org.springframework.http.HttpStatus;
import org.springframework.http.ProblemDetail;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import java.util.NoSuchElementException;
import java.util.stream.Collectors;

/**
 * Traduce los errores del núcleo a respuestas HTTP. Es responsabilidad del
 * adaptador primario, no del caso de uso.
 *
 * Se limita al paquete de este servicio: en un proceso aparte no compite
 * con los manejadores de otros microservicios.
 */
@RestControllerAdvice(basePackages = "co.edu.uptc.farmacia_service.infraestructura.entrada.rest")
public class FarmaciaExceptionHandler {

    @ExceptionHandler(NoSuchElementException.class)
    public ResponseEntity<ProblemDetail> noEncontrado(NoSuchElementException ex) {
        return respuesta(HttpStatus.NOT_FOUND, ex.getMessage());
    }

    @ExceptionHandler(IllegalArgumentException.class)
    public ResponseEntity<ProblemDetail> solicitudInvalida(IllegalArgumentException ex) {
        return respuesta(HttpStatus.BAD_REQUEST, ex.getMessage());
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ProblemDetail> datosInvalidos(MethodArgumentNotValidException ex) {
        String detalle = ex.getBindingResult().getFieldErrors().stream()
                .map(error -> error.getField() + ": " + error.getDefaultMessage())
                .collect(Collectors.joining("; "));
        return respuesta(HttpStatus.BAD_REQUEST, detalle);
    }

    @ExceptionHandler(IllegalStateException.class)
    public ResponseEntity<ProblemDetail> estadoInvalido(IllegalStateException ex) {
        return respuesta(HttpStatus.CONFLICT, ex.getMessage());
    }

    @ExceptionHandler(ServicioExternoNoDisponibleException.class)
    public ResponseEntity<ProblemDetail> servicioExternoNoDisponible(ServicioExternoNoDisponibleException ex) {
        return respuesta(HttpStatus.SERVICE_UNAVAILABLE, ex.getMessage());
    }

    private ResponseEntity<ProblemDetail> respuesta(HttpStatus status, String detalle) {
        return ResponseEntity.status(status)
                .body(ProblemDetail.forStatusAndDetail(status, detalle));
    }
}

import co.edu.uptc.farmacia_service.aplicacion.SolicitudMedicamentoUseCase;
import co.edu.uptc.farmacia_service.dominio.SolicitudMedicamento;
import co.edu.uptc.farmacia_service.infraestructura.entrada.rest.dto.RecibirSolicitudRequest;
import co.edu.uptc.farmacia_service.infraestructura.entrada.rest.dto.SolicitudMedicamentoResponse;
import jakarta.validation.Valid;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import java.util.List;

/**
 * Adaptador primario (driving) de Farmacia y cobertura.
 *
 * Solo depende del puerto primario SolicitudMedicamentoUseCase y del mapper
 * de su propia capa. No conoce la implementación del caso de uso, ni el
 * repositorio, ni la aseguradora.
 */
@RestController
@RequestMapping("/api/farmacia/solicitudes")
public class SolicitudMedicamentoController {

    private final SolicitudMedicamentoUseCase solicitudMedicamentoUseCase;
    private final SolicitudMedicamentoRestMapper mapper;

    public SolicitudMedicamentoController(
            SolicitudMedicamentoUseCase solicitudMedicamentoUseCase,
            SolicitudMedicamentoRestMapper mapper) {
        this.solicitudMedicamentoUseCase = solicitudMedicamentoUseCase;
        this.mapper = mapper;
    }

    @PostMapping
    public ResponseEntity<SolicitudMedicamentoResponse> recibir(@Valid @RequestBody RecibirSolicitudRequest request) {
        SolicitudMedicamento solicitud = solicitudMedicamentoUseCase.recibir(
                request.pacienteId(), request.medicoId(), request.medicamento());
        return ResponseEntity.status(HttpStatus.CREATED).body(mapper.toResponse(solicitud));
    }

    @PostMapping("/{id}/verificar-cobertura")
    public SolicitudMedicamentoResponse verificarCobertura(@PathVariable String id) {
        return mapper.toResponse(solicitudMedicamentoUseCase.verificarCobertura(id));
    }

    @PostMapping("/{id}/despachar")
    public SolicitudMedicamentoResponse despachar(@PathVariable String id) {
        return mapper.toResponse(solicitudMedicamentoUseCase.despachar(id));
    }

    @GetMapping("/{id}")
    public SolicitudMedicamentoResponse buscarPorId(@PathVariable String id) {
        return mapper.toResponse(solicitudMedicamentoUseCase.buscarPorId(id));
    }

    @GetMapping
    public List<SolicitudMedicamentoResponse> listarTodas() {
        return solicitudMedicamentoUseCase.listarTodas().stream()
                .map(mapper::toResponse)
                .toList();
    }
}

package co.edu.uptc.admisiones_service.aplicacion;

import co.edu.uptc.admisiones_service.dominio.DisponibilidadCama;
import co.edu.uptc.admisiones_service.dominio.InventarioCamas;
import org.junit.jupiter.api.Test;

import java.util.*;

import static org.junit.jupiter.api.Assertions.*;

class GestionInventarioCamasServiceTest {

    private final RepositorioInventarioCamasFalso repositorio = new RepositorioInventarioCamasFalso();
    private final GestionInventarioCamasService service = new GestionInventarioCamasService(repositorio);

    @Test
    void registraConsultaOcupaYLiberaUnaCamaSinSpringNiBaseDeDatos() {
        service.registrarCapacidad("sede-tunja", "urgencias", 2);

        DisponibilidadCama inicial = service.consultarDisponibilidad("sede-tunja", "urgencias");
        assertEquals(2, inicial.camasDisponibles());

        service.ocuparCama("sede-tunja", "urgencias");
        service.ocuparCama("sede-tunja", "urgencias");

        assertThrows(IllegalStateException.class, () -> service.ocuparCama("sede-tunja", "urgencias"));

        service.liberarCama("sede-tunja", "urgencias");
        assertEquals(1, service.consultarDisponibilidad("sede-tunja", "urgencias").camasDisponibles());
    }

    @Test
    void fallaSiConsultanUnaSedeServicioNuncaRegistrada() {
        assertThrows(RuntimeException.class, () -> service.consultarDisponibilidad("sede-x", "oncologia"));
    }

    private static class RepositorioInventarioCamasFalso implements RepositorioInventarioCamas {
        private final Map<String, InventarioCamas> almacen = new LinkedHashMap<>();

        @Override
        public Optional<InventarioCamas> buscarPorSedeYServicio(String sedeId, String servicio) {
            return almacen.values().stream()
                    .filter(i -> i.sedeId().equals(sedeId) && i.servicio().equals(servicio))
                    .findFirst();
        }

        @Override
        public List<InventarioCamas> listarPorSede(String sedeId) {
            return almacen.values().stream().filter(i -> i.sedeId().equals(sedeId)).toList();
        }

        @Override
        public InventarioCamas guardar(InventarioCamas inventario) {
            almacen.put(inventario.id(), inventario);
            return inventario;
        }
    }
}

package co.edu.uptc.admisiones_service.dominio;

import jakarta.persistence.*;

import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

@Entity
@Table(name = "admisiones")
public class Admision {

    @Id
    private String id;

    private String pacienteId;
    private String sedeActualId;
    private String servicioActual;

    @Enumerated(EnumType.STRING)
    private EstadoAdmision estado;

    private LocalDateTime fechaIngreso;
    private LocalDateTime fechaAlta;

    @ElementCollection
    @CollectionTable(name = "admision_traslados", joinColumns = @JoinColumn(name = "admision_id"))
    private List<Traslado> historialTraslados = new ArrayList<>();

    protected Admision() {
        // exigido por Hibernate
    }

    public Admision(String id, String pacienteId, String sedeInicialId, String servicioInicial,
            LocalDateTime fechaIngreso) {
        this.id = id;
        this.pacienteId = pacienteId;
        this.sedeActualId = sedeInicialId;
        this.servicioActual = servicioInicial;
        this.estado = EstadoAdmision.ACTIVA;
        this.fechaIngreso = fechaIngreso;
    }

    public void registrarTraslado(String sedeDestinoId, String servicioDestino, LocalDateTime fecha) {
        if (estado != EstadoAdmision.ACTIVA) {
            throw new IllegalStateException("Solo se puede trasladar una admisión activa.");
        }
        historialTraslados.add(new Traslado(
                this.sedeActualId, this.servicioActual, sedeDestinoId, servicioDestino, fecha));
        this.sedeActualId = sedeDestinoId;
        this.servicioActual = servicioDestino;
    }

    public void darDeAlta(boolean tieneOrdenesLaboratorioPendientes, LocalDateTime fecha) {
        if (estado == EstadoAdmision.DADA_DE_ALTA) {
            throw new IllegalStateException("Esta admisión ya fue dada de alta.");
        }
        if (tieneOrdenesLaboratorioPendientes) {
            throw new IllegalStateException(
                    "No se puede dar de alta: el paciente tiene órdenes de laboratorio pendientes.");
        }
        this.estado = EstadoAdmision.DADA_DE_ALTA;
        this.fechaAlta = fecha;
    }

    public String id() {
        return id;
    }

    public String pacienteId() {
        return pacienteId;
    }

    public String sedeActualId() {
        return sedeActualId;
    }

    public String servicioActual() {
        return servicioActual;
    }

    public EstadoAdmision estado() {
        return estado;
    }

    public LocalDateTime fechaIngreso() {
        return fechaIngreso;
    }

    public LocalDateTime fechaAlta() {
        return fechaAlta;
    }

    public List<Traslado> historialTraslados() {

package co.edu.uptc.admisiones_service.dominio;

/**
 * Value Object: DisponibilidadCama.
 *
 * Representa, para una sede y un servicio dados, cuántas camas hay en total
 * y cuántas están ocupadas en este momento (HU-02, HU-10).
 *
 * Es inmutable y no tiene identidad propia: dos DisponibilidadCama con los
 * mismos valores son intercambiables. Se valida en el constructor compacto
 * para que nunca exista una instancia en un estado inconsistente.
 */
public record DisponibilidadCama(
        String sedeId,
        String servicio,
        int camasTotales,
        int camasOcupadas) {

    public DisponibilidadCama {
        if (sedeId == null || sedeId.isBlank()) {
            throw new IllegalArgumentException("La sede es obligatoria para consultar disponibilidad de camas.");
        }
        if (servicio == null || servicio.isBlank()) {
            throw new IllegalArgumentException("El servicio es obligatorio para consultar disponibilidad de camas.");
        }
        if (camasTotales < 0) {
            throw new IllegalArgumentException("El total de camas no puede ser negativo.");
        }
        if (camasOcupadas < 0) {
            throw new IllegalArgumentException("Las camas ocupadas no pueden ser negativas.");
        }
        if (camasOcupadas > camasTotales) {
            throw new IllegalArgumentException(
                    "Las camas ocupadas (%d) no pueden superar el total de camas (%d) de la sede."
                            .formatted(camasOcupadas, camasTotales));
        }
    }

    /** Cuántas camas quedan libres ahora mismo. */
    public int camasDisponibles() {
        return camasTotales - camasOcupadas;
    }

    /**
     * Regla de negocio usada por TrasladoPacienteService y AdmisionFactory:
     * solo se considera "cama confirmada" cuando queda al menos una libre.
     */
    public boolean hayCamaConfirmada() {
        return camasDisponibles() > 0;
    }
}

package co.edu.uptc.admisiones_service.dominio;

import jakarta.persistence.Entity;
import jakarta.persistence.Id;
import jakarta.persistence.Table;
import jakarta.persistence.Version;

/**
 * Raíz del Agregado: InventarioCamas.
 *
 * Es la fuente de verdad de cuántas camas hay y cuántas están ocupadas por
 * sede y servicio (HU-02, HU-10). Antes de este agregado, ese dato lo
 * inventaba quien llamara al endpoint de ingreso — ahora vive aquí y se
 * actualiza atómicamente con cada ingreso, traslado y alta.
 */
@Entity
@Table(name = "inventario_camas")
public class InventarioCamas {

    @Id
    private String id;

    private String sedeId;
    private String servicio;
    private int camasTotales;
    private int camasOcupadas;
    @Version
    private Long version;

    protected InventarioCamas() {
        // exigido por Hibernate
    }

    public InventarioCamas(String id, String sedeId, String servicio, int camasTotales) {
        if (camasTotales < 0) {
            throw new IllegalArgumentException("El total de camas no puede ser negativo.");
        }
        this.id = id;
        this.sedeId = sedeId;
        this.servicio = servicio;
        this.camasTotales = camasTotales;
        this.camasOcupadas = 0;
    }

    public void ocupar() {
        if (camasOcupadas >= camasTotales) {
            throw new IllegalStateException(
                    "No hay cama disponible en sede %s / servicio %s.".formatted(sedeId, servicio));
        }
        camasOcupadas++;
    }

    public void liberar() {
        if (camasOcupadas <= 0) {
            throw new IllegalStateException(
                    "No hay camas ocupadas que liberar en sede %s / servicio %s.".formatted(sedeId, servicio));
        }
        camasOcupadas--;
    }

    public void actualizarCapacidad(int nuevoTotal) {
        if (nuevoTotal < camasOcupadas) {
            throw new IllegalArgumentException(
                    "No se puede fijar un total (%d) menor que las camas ya ocupadas (%d)."
                            .formatted(nuevoTotal, camasOcupadas));
        }
        this.camasTotales = nuevoTotal;
    }

    public DisponibilidadCama aDisponibilidad() {
        return new DisponibilidadCama(sedeId, servicio, camasTotales, camasOcupadas);
    }

    public String id() {
        return id;
    }

    public String sedeId() {
        return sedeId;
    }

    public String servicio() {
        return servicio;
    }
}

            VerificadorLaboratorioGrpcAdapter verificadorLaboratorioGrpc,

        Admision admision = admisionFactory.registrarIngreso(pacienteId, sedeId, disponibilidad);
        Admision guardada = repositorioAdmisiones.guardar(admision);

        inventarioCamasUseCase.ocuparCama(sedeId, servicio);

        String payload = """
                {
                    "admisionId": "%s",
                    "pacienteId": "%s",
                    "sedeInicialId": "%s",
                    "servicioInicial": "%s",
                    "fechaIngreso": "%s"
                }""".formatted(guardada.id(), guardada.pacienteId(), guardada.sedeActualId(),

        String servicioOrigen = admision.servicioActual();
        LocalDateTime fechaTraslado = LocalDateTime.now();

        DisponibilidadCama disponibilidadDestino = inventarioCamasUseCase.consultarDisponibilidad(sedeDestinoId,

        trasladoPacienteService.trasladar(admision, disponibilidadDestino, fechaTraslado);
        repositorioAdmisiones.guardar(admision);

        inventarioCamasUseCase.liberarCama(sedeOrigenId, servicioOrigen);
        inventarioCamasUseCase.ocuparCama(sedeDestinoId, servicioDestino);

        String payload = """

                    "pacienteId": "%s",
                    "sedeOrigenId": "%s",
                    "sedeDestinoId": "%s",
                    "fechaTraslado": "%s"
                }""".formatted(admision.id(), admision.pacienteId(), sedeOrigenId, sedeDestinoId, fechaTraslado);

        outboxRepository.save(new OutboxEvent(admision.pacienteId(), "PacienteTrasladado", payload));
    }

    public void darDeAlta(String admisionId) {

    }

    }

        inventario.ocupar();
        return repositorio.guardar(inventario).aDisponibilidad();
    }

    @Override
    @Transactional
    public void liberarCama(String sedeId, String servicio) {
        InventarioCamas inventario = buscarInventario(sedeId, servicio);
        inventario.liberar();
        repositorio.guardar(inventario);
    }

    private InventarioCamas buscarInventario(String sedeId, String servicio) {
        return repositorio.buscarPorSedeYServicio(sedeId, servicio)
                .orElseThrow(() -> new RecursoNoEncontradoException(
                        "No hay inventario de camas registrado para sede %s / servicio %s. "
                                + "Regístralo primero con POST /api/admisiones/inventario-camas."
                                        .formatted(sedeId, servicio)));
    }

    private String construirId(String sedeId, String servicio) {
        return sedeId + "::" + servicio;
    }
}

import org.slf4j.LoggerFactory;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Transactional;

import java.util.concurrent.TimeUnit;

@Component
public class OutboxPublisher {

    private static final Logger log = LoggerFactory.getLogger(OutboxPublisher.class);
    private final OutboxEventRepository outboxRepository;
    private final KafkaTemplate<String, String> kafkaTemplate;

    public OutboxPublisher(OutboxEventRepository outboxRepository, KafkaTemplate<String, String> kafkaTemplate) {
        this.outboxRepository = outboxRepository;
        this.kafkaTemplate = kafkaTemplate;
    }

    @Scheduled(fixedDelay = 2000)
    @Transactional
    public void publicarPendientes() {
        for (OutboxEvent evento : outboxRepository.findByPublicadoFalseOrderByIdAsc()) {
            try {
                String topico = determinarTopico(evento.getTipo());
                kafkaTemplate.send(topico, evento.getClave(), evento.getPayload())
                        .get(5, TimeUnit.SECONDS);

                evento.marcarPublicado();
                log.info("Outbox: Evento {} ({}) publicado en tópico '{}'", evento.getId(), evento.getTipo(), topico);
            } catch (Exception e) {
                log.warn("Outbox: No se pudo enviar el evento {} (se reintentará): {}", evento.getId(), e.getMessage());
                return; // Mantiene orden estricto interrumpiendo el lote
            }
        }
    }

    private String determinarTopico(String tipoEvento) {
        return switch (tipoEvento) {
            case "PacienteIngresado" -> "admisiones.ingresos-registrados";
            case "PacienteTrasladado" -> "admisiones.pacientes-trasladados";
            case "PacienteDadoDeAlta" -> "admisiones.paciente-dado-alta";
            default -> "admisiones.eventos-generales";
        };
    }
}

                CircuitBreakerConfig.custom()
                        .slidingWindowSize(10)
                        .minimumNumberOfCalls(5)
                        .failureRateThreshold(50)
                        .waitDurationInOpenState(Duration.ofSeconds(10))
                        .build());

                e -> log.warn("Circuit Breaker laboratorio-service: {}", e.getStateTransition()));

        try {
            return circuitBreaker.executeSupplier(() -> {
                var request = OrdenesPendientesRequest.newBuilder()
                        .setPacienteId(pacienteId)
                        .build();
                return stub.withDeadlineAfter(5, TimeUnit.SECONDS)
                        .verificarOrdenesPendientes(request)
                        .getTienePendientes();
            });
        } catch (CallNotPermittedException e) {
            log.warn("Circuit Breaker ABIERTO: laboratorio-service no disponible");
            throw new ServicioNoDisponibleException(
                    "El servicio de laboratorio no está disponible (Circuito Abierto)", e);
        } catch (StatusRuntimeException e) {
            throw new ServicioNoDisponibleException(
                    "No se pudo contactar al servicio de laboratorio via gRPC: " + e.getStatus().getCode(), e);
        }
    }

import java.nio.charset.StandardCharsets;
import java.util.List;

@Configuration
@EnableWebSecurity
public class SecurityConfig {

    @Value("${seguridad.jwt.secret}")
    private String secreto;

    @Bean
    public JwtDecoder jwtDecoder() {
        SecretKeySpec key = new SecretKeySpec(secreto.getBytes(StandardCharsets.UTF_8), "HmacSHA256");
        return NimbusJwtDecoder.withSecretKey(key).build();
    }

    @Bean
    public JwtAuthenticationConverter jwtAuthenticationConverter() {
        JwtAuthenticationConverter converter = new JwtAuthenticationConverter();
        converter.setJwtGrantedAuthoritiesConverter(jwt -> {
            String rol = jwt.getClaimAsString("rol");
            return rol == null ? List.of() : List.of(new SimpleGrantedAuthority("ROLE_" + rol));
        });
        return converter;
    }

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http, ManejadorErroresSeguridad manejador)
            throws Exception {
        http
                .csrf(AbstractHttpConfigurer::disable)
                .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
                .exceptionHandling(e -> e
                        .authenticationEntryPoint(manejador)
                        .accessDeniedHandler(manejador))
                .authorizeHttpRequests(auth -> auth
                        .requestMatchers("/api/status").permitAll()
                        .requestMatchers(HttpMethod.POST, "/api/admisiones/ingreso").hasAuthority("ROLE_ADMISIONES")
                        .requestMatchers(HttpMethod.PUT, "/api/admisiones/*/traslado").hasAuthority("ROLE_MEDICO")
                        .requestMatchers(HttpMethod.PUT, "/api/admisiones/*/alta").hasAuthority("ROLE_MEDICO")
                        .requestMatchers(HttpMethod.POST, "/api/admisiones/inventario-camas")
                        .hasAnyAuthority("ROLE_ADMINISTRADOR", "ROLE_ADMISIONES")
                        .requestMatchers(HttpMethod.GET, "/api/admisiones/inventario-camas/reporte-ocupacion")
                        .hasAuthority("ROLE_ADMINISTRADOR")
                        .requestMatchers(HttpMethod.GET, "/api/admisiones/**")
                        .hasAnyAuthority("ROLE_ADMISIONES", "ROLE_MEDICO", "ROLE_ADMINISTRADOR")
                        .anyRequest().authenticated())
                .oauth2ResourceServer(oauth2 -> oauth2
                        .authenticationEntryPoint(manejador)
                        .accessDeniedHandler(manejador)
                        .jwt(jwt -> jwt.jwtAuthenticationConverter(jwtAuthenticationConverter())));
        return http.build();
    }

package co.edu.uptc.admisiones_service.infraestructura.entrada.controlador.dto;

import co.edu.uptc.admisiones_service.dominio.DisponibilidadCama;

import java.util.List;

public record ReporteOcupacionSede(
        String sedeId,
        int camasTotales,
        int camasOcupadas,
        int camasLibres,
        double porcentajeOcupacion,
        List<DetalleServicio> servicios) {

    public record DetalleServicio(
            String servicio, int camasTotales, int camasOcupadas, int camasLibres, double porcentajeOcupacion) {
    }

    public static ReporteOcupacionSede desde(String sedeId, List<DisponibilidadCama> inventario) {
        List<DetalleServicio> detalle = inventario.stream()
                .map(d -> new DetalleServicio(d.servicio(), d.camasTotales(), d.camasOcupadas(),
                        d.camasTotales() - d.camasOcupadas(), porcentaje(d.camasOcupadas(), d.camasTotales())))
                .toList();

        int totales = detalle.stream().mapToInt(DetalleServicio::camasTotales).sum();
        int ocupadas = detalle.stream().mapToInt(DetalleServicio::camasOcupadas).sum();

        return new ReporteOcupacionSede(sedeId, totales, ocupadas, totales - ocupadas,
                porcentaje(ocupadas, totales), detalle);
    }

    private static double porcentaje(int ocupadas, int totales) {
        return totales == 0 ? 0.0 : Math.round(ocupadas * 1000.0 / totales) / 10.0;
    }
}

package co.edu.uptc.consultas_admisiones_service.dominio.proyeccion;

import co.edu.uptc.consultas_admisiones_service.dominio.modelo.FichaAdmision;
import co.edu.uptc.consultas_admisiones_service.dominio.modelo.ResumenTraslado;
import co.edu.uptc.consultas_admisiones_service.dominio.repositorio.FichaAdmisionRepository;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;

@Service
public class ProyectorFichasAdmision {

    private static final Logger log = LoggerFactory.getLogger(ProyectorFichasAdmision.class);
    private final FichaAdmisionRepository repositorio;
    private final ObjectMapper objectMapper;

    public ProyectorFichasAdmision(FichaAdmisionRepository repositorio, ObjectMapper objectMapper) {
        this.repositorio = repositorio;
        this.objectMapper = objectMapper;
    }

    public synchronized void alIngresarPaciente(String mensajeJson) {
        try {
            JsonNode json = objectMapper.readTree(mensajeJson);
            String admisionId = json.get("admisionId").asText();
            String pacienteId = json.get("pacienteId").asText();
            String sedeInicialId = json.get("sedeInicialId").asText();
            String fechaIngreso = json.get("fechaIngreso").asText();

            FichaAdmision ficha = repositorio.findById(admisionId)
                    .orElseGet(() -> new FichaAdmision(admisionId, pacienteId, sedeInicialId, "ACTIVA", fechaIngreso));

            repositorio.save(ficha);
            log.info("Proyección CQRS: Ficha de admisión {} (Paciente {}) creada/actualizada", admisionId, pacienteId);
        } catch (Exception e) {
            log.error("Error al proyectar el evento PacienteIngresado", e);
        }
    }

    public synchronized void alTrasladarPaciente(String mensajeJson) {
        try {
            JsonNode json = objectMapper.readTree(mensajeJson);
            String admisionId = json.get("admisionId").asText();
            String sedeOrigenId = json.get("sedeOrigenId").asText();
            String sedeDestinoId = json.get("sedeDestinoId").asText();
            String fechaTraslado = json.get("fechaTraslado").asText();

            FichaAdmision ficha = repositorio.findById(admisionId).orElse(null);
            if (ficha != null) {
                ResumenTraslado traslado = new ResumenTraslado(sedeOrigenId, sedeDestinoId, fechaTraslado);
                boolean agregado = ficha.agregarTraslado(traslado);
                if (agregado) {
                    repositorio.save(ficha);
                    log.info("Proyección CQRS: Traslado registrado en admisión {} a la sede {}", admisionId,
                            sedeDestinoId);
                } else {
                    log.info("Proyección CQRS: Traslado duplicado ignorado para admisión {}", admisionId);
                }
            } else {
                log.warn("Proyección CQRS: Traslado recibido antes de crear la admisión {}", admisionId);
            }
        } catch (Exception e) {
            log.error("Error al proyectar el evento PacienteTrasladado", e);
        }
    }

    public synchronized void alDarDeAltaPaciente(String mensajeJson) {
        try {
            JsonNode json = objectMapper.readTree(mensajeJson);
            String admisionId = json.get("admisionId").asText();
            String fechaAlta = json.get("fechaAlta").asText();

            FichaAdmision ficha = repositorio.findById(admisionId).orElse(null);
            if (ficha != null) {
                ficha.darDeAlta(fechaAlta);
                repositorio.save(ficha);
                log.info("Proyección CQRS: Admisión {} dada de alta en el modelo de lectura", admisionId);
            } else {
                log.warn("Proyección CQRS: Alta recibida para admisión no registrada {}", admisionId);
            }
        } catch (Exception e) {
            log.error("Error al proyectar el evento PacienteDadoDeAlta", e);
        }
    }
}

package co.edu.uptc.consultas_admisiones_service.dominio.proyeccion;

import co.edu.uptc.consultas_admisiones_service.dominio.modelo.FichaMedicacion;
import co.edu.uptc.consultas_admisiones_service.dominio.repositorio.FichaMedicacionRepository;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;

@Service
public class ProyectorFichasMedicacion {

    private static final Logger log = LoggerFactory.getLogger(ProyectorFichasMedicacion.class);
    private final FichaMedicacionRepository repositorio;
    private final ObjectMapper objectMapper;

    public ProyectorFichasMedicacion(FichaMedicacionRepository repositorio, ObjectMapper objectMapper) {
        this.repositorio = repositorio;
        this.objectMapper = objectMapper;
    }

    // farmacia-service publica los 3 eventos (MedicamentoSolicitado,
    // CoberturaVerificada, MedicamentoDespachado) en UN SOLO tópico
    // ("farmacia.solicitudes"), diferenciados por el campo "tipo" dentro
    // del JSON — a diferencia de admisiones, que usa un tópico por evento.
    // Por eso aquí no hace falta un método por tipo de evento: basta con
    // upsert sobre el mismo documento por solicitudId.
    public synchronized void alRecibirEventoFarmacia(String mensajeJson) {
        try {
            JsonNode json = objectMapper.readTree(mensajeJson);
            String tipo = json.get("tipo").asText();
            String solicitudId = json.get("solicitudId").asText();
            String pacienteId = json.get("pacienteId").asText();

            FichaMedicacion ficha = repositorio.findById(solicitudId)
                    .orElseGet(() -> new FichaMedicacion(solicitudId, pacienteId));

            ficha.setMedicoId(textoONulo(json, "medicoId"));
            ficha.setMedicamento(textoONulo(json, "medicamento"));
            ficha.setEstado(textoONulo(json, "estado"));
            ficha.setFechaSolicitud(textoONulo(json, "fechaSolicitud"));
            ficha.setFechaDespacho(textoONulo(json, "fechaDespacho"));
            if (json.hasNonNull("cobertura")) {
                ficha.setCoberturaPorcentaje(json.get("cobertura").asInt());
            }

            repositorio.save(ficha);
            log.info("Proyección CQRS: ficha de medicación {} ({}) actualizada para paciente {}",
                    solicitudId, tipo, pacienteId);
        } catch (Exception e) {
            log.error("Error al proyectar un evento de farmacia-service", e);
        }
    }

    private String textoONulo(JsonNode json, String campo) {
        return json.hasNonNull(campo) ? json.get(campo).asText() : null;
    }
}

public class FichaAdmision {

    @Id
    private String admisionId;
    private String pacienteId;
    private String sedeActualId;
    private String estado; // ACTIVA, DADA_DE_ALTA
    private String fechaIngreso;
    private String fechaAlta;
    private List<ResumenTraslado> traslados = new ArrayList<>();
    private int totalTraslados;

    public FichaAdmision() {
    }

    public FichaAdmision(String admisionId, String pacienteId, String sedeActualId, String estado,
            String fechaIngreso) {
        this.admisionId = admisionId;
        this.pacienteId = pacienteId;
        this.sedeActualId = sedeActualId;
        this.estado = estado;
        this.fechaIngreso = fechaIngreso;
        this.totalTraslados = 0;
    }

    public boolean agregarTraslado(ResumenTraslado traslado) {
        boolean duplicado = traslados.stream()
                .anyMatch(t -> t.sedeOrigenId().equals(traslado.sedeOrigenId()) &&
                        t.sedeDestinoId().equals(traslado.sedeDestinoId()) &&
                        t.fecha().equals(traslado.fecha()));
        if (duplicado) {
            return false;
        }
        this.traslados.add(traslado);
        this.sedeActualId = traslado.sedeDestinoId();
        this.totalTraslados = this.traslados.size();
        return true;
    }

    public void darDeAlta(String fechaAlta) {
        this.estado = "DADA_DE_ALTA";
        this.fechaAlta = fechaAlta;
    }

        String mensaje = ex.getBindingResult().getFieldErrors().stream()
                .map(FieldError::getDefaultMessage)
                .reduce((a, b) -> a + "; " + b)
                .orElse("Datos inválidos.");
        return construirRespuesta(HttpStatus.BAD_REQUEST, mensaje);
    }

    @ExceptionHandler(MissingServletRequestParameterException.class)
    public ResponseEntity<Map<String, Object>> manejarParametroFaltante(MissingServletRequestParameterException ex) {
        return construirRespuesta(HttpStatus.BAD_REQUEST, "Falta el parámetro obligatorio: " + ex.getParameterName());
    }

    @ExceptionHandler(MethodArgumentTypeMismatchException.class)
    public ResponseEntity<Map<String, Object>> manejarTipoInvalido(MethodArgumentTypeMismatchException ex) {
        return construirRespuesta(HttpStatus.BAD_REQUEST,
                "El parámetro '%s' tiene un formato inválido.".formatted(ex.getName()));
    }

    @ExceptionHandler(HttpMessageNotReadableException.class)
    public ResponseEntity<Map<String, Object>> manejarJsonInvalido(HttpMessageNotReadableException ex) {
        return construirRespuesta(HttpStatus.BAD_REQUEST, "El cuerpo de la petición no es un JSON válido.");
    }

    @ExceptionHandler(OptimisticLockingFailureException.class)
    public ResponseEntity<Map<String, Object>> manejarConflictoConcurrencia(
            OptimisticLockingFailureException ex) {
        return construirRespuesta(HttpStatus.CONFLICT,
                "El inventario de camas cambió mientras procesábamos tu solicitud. Intenta de nuevo.");
    }

    // --- catch-all, ahora sí al final de verdad ---

    @ExceptionHandler(Exception.class)
    public ResponseEntity<Map<String, Object>> manejarExcepcionGeneral(Exception ex) {
        return construirRespuesta(HttpStatus.INTERNAL_SERVER_ERROR, "Error interno del servidor: " + ex.getMessage());
    }

    private ResponseEntity<Map<String, Object>> construirRespuesta(HttpStatus status, String mensaje) {
        Map<String, Object> cuerpo = new LinkedHashMap<>();
        cuerpo.put("timestamp", LocalDateTime.now().toString());
        cuerpo.put("status", status.value());
        cuerpo.put("error", status.getReasonPhrase());
        cuerpo.put("mensaje", mensaje);
        return ResponseEntity.status(status).body(cuerpo);
    }
}

package co.edu.uptc.consultas_admisiones_service.infraestructura.seguridad; // ← ajusta por servicio

import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.http.HttpStatus;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.security.core.AuthenticationException;
import org.springframework.security.web.AuthenticationEntryPoint;
import org.springframework.security.web.access.AccessDeniedHandler;
import org.springframework.stereotype.Component;

import java.io.IOException;
import java.time.LocalDateTime;

@Component
public class ManejadorErroresSeguridad implements AuthenticationEntryPoint, AccessDeniedHandler {

    @Override
    public void commence(HttpServletRequest request, HttpServletResponse response,
            AuthenticationException ex) throws IOException {
        response.setHeader("WWW-Authenticate", "Bearer");
        escribir(response, HttpStatus.UNAUTHORIZED,
                "Token ausente, inválido o expirado. Inicia sesión en /api/auth/login.");
    }

    @Override
    public void handle(HttpServletRequest request, HttpServletResponse response,
            AccessDeniedException ex) throws IOException {
        escribir(response, HttpStatus.FORBIDDEN, "Tu rol no tiene permiso para realizar esta acción.");
    }

    private void escribir(HttpServletResponse response, HttpStatus status, String mensaje) throws IOException {
        response.setStatus(status.value());
        response.setContentType("application/json;charset=UTF-8");
        response.getWriter().write(
                "{\"timestamp\":\"%s\",\"status\":%d,\"error\":\"%s\",\"mensaje\":\"%s\"}"
                        .formatted(LocalDateTime.now(), status.value(), status.getReasonPhrase(), mensaje));
    }
}

package co.edu.uptc.consultas_admisiones_service.infraestructura.seguridad;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpMethod;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.annotation.web.configurers.AbstractHttpConfigurer;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.oauth2.jwt.JwtDecoder;
import org.springframework.security.oauth2.jwt.NimbusJwtDecoder;
import org.springframework.security.oauth2.server.resource.authentication.JwtAuthenticationConverter;
import org.springframework.security.web.SecurityFilterChain;

import javax.crypto.spec.SecretKeySpec;
import java.nio.charset.StandardCharsets;
import java.util.List;

@Configuration
@EnableWebSecurity
public class SecurityConfig {

    @Value("${seguridad.jwt.secret}")
    private String secreto;

    @Bean
    public JwtDecoder jwtDecoder() {
        SecretKeySpec key = new SecretKeySpec(secreto.getBytes(StandardCharsets.UTF_8), "HmacSHA256");
        return NimbusJwtDecoder.withSecretKey(key).build();
    }

    @Bean
    public JwtAuthenticationConverter jwtAuthenticationConverter() {
        JwtAuthenticationConverter converter = new JwtAuthenticationConverter();
        converter.setJwtGrantedAuthoritiesConverter(jwt -> {
            String rol = jwt.getClaimAsString("rol");
            return rol == null ? List.of() : List.of(new SimpleGrantedAuthority("ROLE_" + rol));
        });
        return converter;
    }

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http, ManejadorErroresSeguridad manejador)
            throws Exception {
        http
                .csrf(AbstractHttpConfigurer::disable)
                .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
                .exceptionHandling(e -> e
                        .authenticationEntryPoint(manejador)
                        .accessDeniedHandler(manejador))
                .authorizeHttpRequests(auth -> auth
                        .requestMatchers("/api/status").permitAll()
                        .requestMatchers(HttpMethod.GET, "/api/consultas/admisiones/reporte/**")
                        .hasAuthority("ROLE_ADMINISTRADOR")
                        .requestMatchers(HttpMethod.GET, "/api/consultas/admisiones/**")
                        .hasAnyAuthority("ROLE_MEDICO", "ROLE_ADMISIONES", "ROLE_ADMINISTRADOR")
                        .anyRequest().authenticated())
                .oauth2ResourceServer(oauth2 -> oauth2
                        .authenticationEntryPoint(manejador)
                        .accessDeniedHandler(manejador)
                        .jwt(jwt -> jwt.jwtAuthenticationConverter(jwtAuthenticationConverter())));
        return http.build();
    }
}

package co.edu.uptc.consultas_admisiones_service.infraestructura.entrada.controlador;

import co.edu.uptc.consultas_admisiones_service.compartido.RecursoNoEncontradoException;

import co.edu.uptc.consultas_admisiones_service.dominio.repositorio.FichaMedicacionRepository;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.List;
import java.util.Map;

@RestController
@RequestMapping("/api/consultas/admisiones")
public class ConsultaAdmisionController {

    private final FichaAdmisionRepository fichaAdmisionRepository;
    private final FichaMedicacionRepository fichaMedicacionRepository;

    public ConsultaAdmisionController(
            FichaAdmisionRepository fichaAdmisionRepository,
            FichaMedicacionRepository fichaMedicacionRepository) {
        this.fichaAdmisionRepository = fichaAdmisionRepository;
        this.fichaMedicacionRepository = fichaMedicacionRepository;
    }

    @GetMapping
    public ResponseEntity<List<FichaAdmision>> listarTodas(@RequestParam(required = false) String estado) {
        if (estado != null) {
            return ResponseEntity.ok(fichaAdmisionRepository.findByEstado(estado.toUpperCase()));
        }
        return ResponseEntity.ok(fichaAdmisionRepository.findAll());
    }

    @GetMapping("/{admisionId}")
    public ResponseEntity<FichaAdmision> buscarPorId(@PathVariable String admisionId) {
        return fichaAdmisionRepository.findById(admisionId)
                .map(ResponseEntity::ok)
                .orElseThrow(
                        () -> new RecursoNoEncontradoException("No existe ficha de admisión con ID: " + admisionId));
    }

    @GetMapping("/paciente/{pacienteId}")
    public ResponseEntity<List<FichaAdmision>> buscarPorPaciente(@PathVariable String pacienteId) {
        return ResponseEntity.ok(fichaAdmisionRepository.findByPacienteId(pacienteId));
    }

    /**
     * HU-08: estado consolidado del paciente (admisión + medicación; laboratorio se
     * suma después).
     */
    @GetMapping("/paciente/{pacienteId}/consolidado")
    public ResponseEntity<EstadoConsolidadoPaciente> consultarEstadoConsolidado(@PathVariable String pacienteId) {
        var admisiones = fichaAdmisionRepository.findByPacienteId(pacienteId);
        var medicaciones = fichaMedicacionRepository.findByPacienteId(pacienteId);
        return ResponseEntity.ok(new EstadoConsolidadoPaciente(pacienteId, admisiones, medicaciones));
    }

    @GetMapping("/reporte/ocupacion")
    public ResponseEntity<Map<String, Object>> reporteOcupacion(@RequestParam String sedeId) {
        var activas = fichaAdmisionRepository.findBySedeActualIdAndEstado(sedeId, "ACTIVA");
        return ResponseEntity.ok(Map.of(
                "sedeId", sedeId,
                "camasOcupadasEnTiempoReal", activas.size(),
                "pacientesAdmitidos", activas));
    }
}

import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import java.time.LocalDateTime;
import java.util.LinkedHashMap;
import java.util.Map;

@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(CorreoYaRegistradoException.class)
    public ResponseEntity<Map<String, Object>> manejarCorreoDuplicado(CorreoYaRegistradoException ex) {
        return construir(HttpStatus.CONFLICT, ex.getMessage());
    }

    @ExceptionHandler(CredencialesInvalidasException.class)
    public ResponseEntity<Map<String, Object>> manejarCredencialesInvalidas(CredencialesInvalidasException ex) {
        return construir(HttpStatus.UNAUTHORIZED, ex.getMessage());
    }

    @ExceptionHandler(IllegalArgumentException.class)
    public ResponseEntity<Map<String, Object>> manejarIllegalArgument(IllegalArgumentException ex) {
        return construir(HttpStatus.BAD_REQUEST, ex.getMessage());
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<Map<String, Object>> manejarValidacion(MethodArgumentNotValidException ex) {
        String mensaje = ex.getBindingResult().getFieldErrors().stream()
                .map(e -> e.getField() + ": " + e.getDefaultMessage())
                .reduce((a, b) -> a + "; " + b)
                .orElse("Datos inválidos.");
        return construir(HttpStatus.BAD_REQUEST, mensaje);
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<Map<String, Object>> manejarGeneral(Exception ex) {
        return construir(HttpStatus.INTERNAL_SERVER_ERROR, "Error interno del servidor: " + ex.getMessage());
    }

    private ResponseEntity<Map<String, Object>> construir(HttpStatus status, String mensaje) {
        Map<String, Object> cuerpo = new LinkedHashMap<>();
        cuerpo.put("timestamp", LocalDateTime.now().toString());
        cuerpo.put("status", status.value());
        cuerpo.put("error", status.getReasonPhrase());
        cuerpo.put("mensaje", mensaje);
        return ResponseEntity.status(status).body(cuerpo);
    }
}

package co.edu.uptc.auth_service.aplicacion;

import co.edu.uptc.auth_service.compartido.CorreoYaRegistradoException;
import co.edu.uptc.auth_service.dominio.Rol;
import co.edu.uptc.auth_service.dominio.Usuario;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;

import java.util.UUID;

@Service
public class RegistroUsuarioService {

    private final RepositorioUsuarios repositorio;
    private final PasswordEncoder passwordEncoder;

    public RegistroUsuarioService(RepositorioUsuarios repositorio, PasswordEncoder passwordEncoder) {
        this.repositorio = repositorio;
        this.passwordEncoder = passwordEncoder;
    }

    public Usuario registrar(String email, String password, Rol rol) {
        if (email == null || email.isBlank()) {
            throw new IllegalArgumentException("El correo es obligatorio.");
        }
        if (password == null || password.length() < 8) {
            throw new IllegalArgumentException("La contraseña debe tener al menos 8 caracteres.");
        }
        if (rol == null) {
            throw new IllegalArgumentException("El rol es obligatorio.");
        }
        if (repositorio.existePorEmail(email)) {
            throw new CorreoYaRegistradoException("Ya existe un usuario registrado con el correo " + email);
        }

        Usuario usuario = new Usuario(UUID.randomUUID().toString(), email, passwordEncoder.encode(password), rol);
        return repositorio.guardar(usuario);
    }
}

package co.edu.uptc.auth_service.aplicacion;

import co.edu.uptc.auth_service.compartido.CredencialesInvalidasException;
import co.edu.uptc.auth_service.dominio.Usuario;
import co.edu.uptc.auth_service.infraestructura.seguridad.JwtService;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;

@Service
public class AutenticacionService {

    private final RepositorioUsuarios repositorio;
    private final PasswordEncoder passwordEncoder;
    private final JwtService jwtService;

    public AutenticacionService(RepositorioUsuarios repositorio, PasswordEncoder passwordEncoder,
            JwtService jwtService) {
        this.repositorio = repositorio;
        this.passwordEncoder = passwordEncoder;
        this.jwtService = jwtService;
    }

    public ResultadoAutenticacion autenticar(String email, String password) {
        Usuario usuario = repositorio.buscarPorEmail(email)
                .orElseThrow(() -> new CredencialesInvalidasException("Correo o contraseña incorrectos."));

        if (!passwordEncoder.matches(password, usuario.passwordHash())) {
            throw new CredencialesInvalidasException("Correo o contraseña incorrectos.");
        }

        String token = jwtService.generar(usuario.email(), usuario.rol());
        return new ResultadoAutenticacion(token, usuario.rol().name(), jwtService.expiracionSegundos());
    }
}

import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.http.HttpStatus;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.security.core.AuthenticationException;
import org.springframework.security.web.AuthenticationEntryPoint;
import org.springframework.security.web.access.AccessDeniedHandler;
import org.springframework.stereotype.Component;

import java.io.IOException;
import java.time.LocalDateTime;

@Component
public class ManejadorErroresSeguridad implements AuthenticationEntryPoint, AccessDeniedHandler {

    @Override
    public void commence(HttpServletRequest request, HttpServletResponse response,
            AuthenticationException ex) throws IOException {
        response.setHeader("WWW-Authenticate", "Bearer");
        escribir(response, HttpStatus.UNAUTHORIZED,
                "Token ausente, inválido o expirado. Inicia sesión en /api/auth/login.");
    }

    @Override
    public void handle(HttpServletRequest request, HttpServletResponse response,
            AccessDeniedException ex) throws IOException {
        escribir(response, HttpStatus.FORBIDDEN, "Tu rol no tiene permiso para realizar esta acción.");
    }

    private void escribir(HttpServletResponse response, HttpStatus status, String mensaje) throws IOException {
        response.setStatus(status.value());
        response.setContentType("application/json;charset=UTF-8");
        response.getWriter().write(
                "{\"timestamp\":\"%s\",\"status\":%d,\"error\":\"%s\",\"mensaje\":\"%s\"}"
                        .formatted(LocalDateTime.now(), status.value(), status.getReasonPhrase(), mensaje));
    }
}

package co.edu.uptc.auth_service.infraestructura.seguridad;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpMethod;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.annotation.web.configurers.AbstractHttpConfigurer;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.oauth2.jwt.JwtDecoder;
import org.springframework.security.oauth2.jwt.NimbusJwtDecoder;
import org.springframework.security.oauth2.server.resource.authentication.JwtAuthenticationConverter;
import org.springframework.security.web.SecurityFilterChain;

import javax.crypto.spec.SecretKeySpec;
import java.nio.charset.StandardCharsets;
import java.util.List;

@Configuration
@EnableWebSecurity
public class SecurityConfig {

    @Value("${seguridad.jwt.secret}")
    private String secreto;

    @Bean
    public JwtDecoder jwtDecoder() {
        SecretKeySpec key = new SecretKeySpec(secreto.getBytes(StandardCharsets.UTF_8), "HmacSHA256");
        return NimbusJwtDecoder.withSecretKey(key).build();
    }

    @Bean
    public JwtAuthenticationConverter jwtAuthenticationConverter() {
        JwtAuthenticationConverter converter = new JwtAuthenticationConverter();
        converter.setJwtGrantedAuthoritiesConverter(jwt -> {
            String rol = jwt.getClaimAsString("rol");
            return rol == null ? List.of() : List.of(new SimpleGrantedAuthority("ROLE_" + rol));
        });
        return converter;
    }

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http, ManejadorErroresSeguridad manejador)
            throws Exception {
        http
                .csrf(AbstractHttpConfigurer::disable)
                .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
                .exceptionHandling(e -> e
                        .authenticationEntryPoint(manejador)
                        .accessDeniedHandler(manejador))
                .authorizeHttpRequests(auth -> auth
                        .requestMatchers(HttpMethod.POST, "/api/auth/login").permitAll()
                        .requestMatchers(HttpMethod.POST, "/api/auth/usuarios").hasAuthority("ROLE_ADMINISTRADOR")
                        .anyRequest().authenticated())
                .oauth2ResourceServer(oauth2 -> oauth2
                        .authenticationEntryPoint(manejador)
                        .accessDeniedHandler(manejador)
                        .jwt(jwt -> jwt.jwtAuthenticationConverter(jwtAuthenticationConverter())));
        return http.build();
    }
}

import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.time.Instant;
import java.util.Date;

@Component
public class JwtService {

    private final SecretKey key;
    private final long expiracionMs;

    public JwtService(
            @Value("${seguridad.jwt.secret}") String secreto,
            @Value("${seguridad.jwt.expiracion-ms:7200000}") long expiracionMs) {
        this.key = Keys.hmacShaKeyFor(secreto.getBytes(StandardCharsets.UTF_8));
        this.expiracionMs = expiracionMs;
    }

    public String generar(String email, Rol rol) {
        Instant ahora = Instant.now();
        return Jwts.builder()
                .subject(email)
                .claim("rol", rol.name())
                .issuedAt(Date.from(ahora))
                .expiration(Date.from(ahora.plusMillis(expiracionMs)))
                .signWith(key)
                .compact();
    }

    public long expiracionSegundos() {
        return expiracionMs / 1000;
    }
}

package co.edu.uptc.auth_service.infraestructura.entrada.rest;

import co.edu.uptc.auth_service.aplicacion.AutenticacionService;
import co.edu.uptc.auth_service.aplicacion.RegistroUsuarioService;
import co.edu.uptc.auth_service.aplicacion.ResultadoAutenticacion;
import co.edu.uptc.auth_service.dominio.Usuario;
import co.edu.uptc.auth_service.infraestructura.entrada.rest.dto.*;
import jakarta.validation.Valid;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api/auth")
public class AuthController {

    private final RegistroUsuarioService registroUsuarioService;
    private final AutenticacionService autenticacionService;

    public AuthController(RegistroUsuarioService registroUsuarioService, AutenticacionService autenticacionService) {
        this.registroUsuarioService = registroUsuarioService;
        this.autenticacionService = autenticacionService;
    }

    /** Público: la única puerta de entrada sin credenciales previas. */
    @PostMapping("/login")
    public ResponseEntity<LoginResponse> login(@Valid @RequestBody LoginRequest request) {
        ResultadoAutenticacion resultado = autenticacionService.autenticar(request.email(), request.password());
        return ResponseEntity.ok(new LoginResponse(resultado.token(), resultado.rol(), resultado.expiraEnSegundos()));
    }

    /**
     * Protegido: solo un ADMINISTRADOR ya autenticado puede crear cuentas
     * de personal (médico, admisiones, laboratorio, farmacéutico, u otro
     * administrador). Esto reemplaza al antiguo "registro público".
     */
    @PostMapping("/usuarios")
    public ResponseEntity<UsuarioResponse> crearUsuario(@Valid @RequestBody RegistroRequest request) {
        Usuario usuario = registroUsuarioService.registrar(request.email(), request.password(), request.rol());
        return ResponseEntity.status(HttpStatus.CREATED)
                .body(new UsuarioResponse(usuario.id(), usuario.email(), usuario.rol().name()));
    }
}

package co.edu.uptc.auth_service.infraestructura.arranque;

import co.edu.uptc.auth_service.aplicacion.RepositorioUsuarios;
import co.edu.uptc.auth_service.dominio.Rol;
import co.edu.uptc.auth_service.dominio.Usuario;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.ApplicationArguments;
import org.springframework.boot.ApplicationRunner;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Component;

import java.util.UUID;

/**
 * Crea el primer administrador del sistema la primera vez que el servicio
 * arranca y no existe ninguno todavía. Después de eso, cualquier otra
 * cuenta (médico, laboratorio, farmacéutico, admisiones, u OTROS
 * administradores) se crea desde /api/auth/usuarios, autenticado como
 * ADMINISTRADOR — nunca por registro público.
 */
@Component
public class SembradorAdministradorInicial implements ApplicationRunner {

    private static final Logger log = LoggerFactory.getLogger(SembradorAdministradorInicial.class);

    private final RepositorioUsuarios repositorio;
    private final PasswordEncoder passwordEncoder;
    private final String emailAdmin;
    private final String passwordAdmin;

    public SembradorAdministradorInicial(
            RepositorioUsuarios repositorio,
            PasswordEncoder passwordEncoder,
            @Value("${seguridad.admin-inicial.email:admin@hospital.com}") String emailAdmin,
            @Value("${seguridad.admin-inicial.password:CambiaEstaClave123!}") String passwordAdmin) {
        this.repositorio = repositorio;
        this.passwordEncoder = passwordEncoder;
        this.emailAdmin = emailAdmin;
        this.passwordAdmin = passwordAdmin;
    }

    @Override
    public void run(ApplicationArguments args) {
        if (repositorio.buscarPorEmail(emailAdmin).isPresent()) {
            return; // ya existe, no se repite en próximos arranques
        }

        Usuario admin = new Usuario(
                UUID.randomUUID().toString(), emailAdmin, passwordEncoder.encode(passwordAdmin), Rol.ADMINISTRADOR);
        repositorio.guardar(admin);

        log.warn("Administrador inicial creado automáticamente: {} — "
                + "cámbiale la contraseña por defecto antes de usar esto fuera de desarrollo.", emailAdmin);
    }
}

package co.edu.uptc.gateway.seguridad;

import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatus;
import org.springframework.http.MediaType;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.security.core.AuthenticationException;
import org.springframework.security.web.server.ServerAuthenticationEntryPoint;
import org.springframework.security.web.server.authorization.ServerAccessDeniedHandler;
import org.springframework.stereotype.Component;
import org.springframework.web.server.ServerWebExchange;
import reactor.core.publisher.Mono;

import java.nio.charset.StandardCharsets;
import java.time.LocalDateTime;

@Component
public class ManejadorErroresSeguridad implements ServerAuthenticationEntryPoint, ServerAccessDeniedHandler {

    @Override
    public Mono<Void> commence(ServerWebExchange exchange, AuthenticationException ex) {
        exchange.getResponse().getHeaders().set(HttpHeaders.WWW_AUTHENTICATE, "Bearer");
        return escribir(exchange, HttpStatus.UNAUTHORIZED,
                "Token ausente, inválido o expirado. Inicia sesión en /api/auth/login.");
    }

    @Override
    public Mono<Void> handle(ServerWebExchange exchange, AccessDeniedException ex) {
        return escribir(exchange, HttpStatus.FORBIDDEN, "Tu rol no tiene permiso para realizar esta acción.");
    }

    private Mono<Void> escribir(ServerWebExchange exchange, HttpStatus status, String mensaje) {
        var response = exchange.getResponse();
        response.setStatusCode(status);
        response.getHeaders().setContentType(MediaType.APPLICATION_JSON);
        byte[] cuerpo = "{\"timestamp\":\"%s\",\"status\":%d,\"error\":\"%s\",\"mensaje\":\"%s\"}"
                .formatted(LocalDateTime.now(), status.value(), status.getReasonPhrase(), mensaje)
                .getBytes(StandardCharsets.UTF_8);
        return response.writeWith(Mono.just(response.bufferFactory().wrap(cuerpo)));
    }
}

package co.edu.uptc.gateway.seguridad;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpMethod;
import org.springframework.security.config.annotation.web.reactive.EnableWebFluxSecurity;
import org.springframework.security.config.web.server.ServerHttpSecurity;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.oauth2.jwt.NimbusReactiveJwtDecoder;
import org.springframework.security.oauth2.jwt.ReactiveJwtDecoder;
import org.springframework.security.oauth2.server.resource.authentication.ReactiveJwtAuthenticationConverter;
import org.springframework.security.web.server.SecurityWebFilterChain;
import reactor.core.publisher.Flux;

import javax.crypto.spec.SecretKeySpec;
import java.nio.charset.StandardCharsets;

@Configuration
@EnableWebFluxSecurity
public class SecurityConfig {

    @Value("${seguridad.jwt.secret}")
    private String secreto;

    @Bean
    public ReactiveJwtDecoder jwtDecoder() {
        SecretKeySpec key = new SecretKeySpec(secreto.getBytes(StandardCharsets.UTF_8), "HmacSHA256");
        return NimbusReactiveJwtDecoder.withSecretKey(key).build();
    }

    @Bean
    public ReactiveJwtAuthenticationConverter jwtAuthenticationConverter() {
        var converter = new ReactiveJwtAuthenticationConverter();
        converter.setJwtGrantedAuthoritiesConverter(jwt -> {
            String rol = jwt.getClaimAsString("rol");
            return rol == null ? Flux.empty() : Flux.just(new SimpleGrantedAuthority("ROLE_" + rol));
        });
        return converter;
    }

    @Bean
    public SecurityWebFilterChain securityWebFilterChain(ServerHttpSecurity http,
            ManejadorErroresSeguridad manejador) {
        http
                .csrf(ServerHttpSecurity.CsrfSpec::disable)
                .exceptionHandling(e -> e
                        .authenticationEntryPoint(manejador)
                        .accessDeniedHandler(manejador))
                .authorizeExchange(exchanges -> exchanges
                        .pathMatchers(HttpMethod.POST, "/api/auth/login").permitAll()
                        .pathMatchers(HttpMethod.POST, "/api/auth/usuarios").hasAuthority("ROLE_ADMINISTRADOR")
                        .pathMatchers(HttpMethod.POST, "/api/admisiones/ingreso").hasAuthority("ROLE_ADMISIONES")
                        .pathMatchers(HttpMethod.PUT, "/api/admisiones/*/traslado").hasAuthority("ROLE_MEDICO")
                        .pathMatchers(HttpMethod.PUT, "/api/admisiones/*/alta").hasAuthority("ROLE_MEDICO")
                        .pathMatchers(HttpMethod.POST, "/api/admisiones/inventario-camas")
                        .hasAnyAuthority("ROLE_ADMINISTRADOR", "ROLE_ADMISIONES")
                        .pathMatchers(HttpMethod.GET, "/api/admisiones/inventario-camas/reporte-ocupacion")
                        .hasAuthority("ROLE_ADMINISTRADOR")
                        .pathMatchers(HttpMethod.GET, "/api/admisiones/**")
                        .hasAnyAuthority("ROLE_ADMISIONES", "ROLE_MEDICO", "ROLE_ADMINISTRADOR")
                        .pathMatchers(HttpMethod.GET, "/api/consultas/admisiones/reporte/**")
                        .hasAuthority("ROLE_ADMINISTRADOR")
                        .pathMatchers("/api/consultas/admisiones/**")
                        .hasAnyAuthority("ROLE_MEDICO", "ROLE_ADMISIONES", "ROLE_ADMINISTRADOR")
                        .pathMatchers(HttpMethod.POST, "/api/farmacia/solicitudes").hasAuthority("ROLE_FARMACEUTICO")
                        .pathMatchers(HttpMethod.POST, "/api/farmacia/solicitudes/*/despachar")
                        .hasAuthority("ROLE_FARMACEUTICO")
                        .pathMatchers("/api/farmacia/**")
                        .hasAnyAuthority("ROLE_FARMACEUTICO", "ROLE_ADMISIONES", "ROLE_MEDICO")
                        .pathMatchers("/api/laboratorio/**").hasAnyAuthority("ROLE_MEDICO", "ROLE_LABORATORIO")
                        .anyExchange().authenticated())
                .oauth2ResourceServer(oauth2 -> oauth2
                        .authenticationEntryPoint(manejador)
                        .accessDeniedHandler(manejador)
                        .jwt(jwt -> jwt.jwtAuthenticationConverter(jwtAuthenticationConverter())));
        return http.build();
    }
}
